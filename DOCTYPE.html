    <script>

        const chatContainer = document.getElementById('chat-container');

        const messageInput = document.getElementById('message-input');

        const usernameInput = document.getElementById('username-input');

 

        let messagesList = [];

 

        function initChat() {

            const saved = localStorage.getItem('zchat_messages');

            if (saved) {

                messagesList = JSON.parse(saved);

            } else {

                messagesList = [

                    { sender: "Sistema", text: "¡Bienvenido a ZCHAT! Envía tu primer mensaje.", timestamp: Date.now() }

                ];

            }

            renderMessages();

        }

 

        function sendMessage() {

            const text = messageInput.value.trim();

            const sender = usernameInput.value.trim() || "Anónimo";

 

            if (!text) return;

 

            const newMessage = {

                sender: sender,

                text: text,

                timestamp: Date.now()

            };

 

            messagesList.push(newMessage);

            localStorage.setItem('zchat_messages', JSON.stringify(messagesList));

            renderMessages();

 

            messageInput.value = "";

            messageInput.focus();

        }

 

        function renderMessages() {

            chatContainer.innerHTML = "";

            const currentUser = usernameInput.value.trim();

 

            messagesList.forEach((msg) => {

                const div = document.createElement('div');

                const isMe = msg.sender === currentUser;

                div.className = `message ${isMe ? 'me' : 'other'}`;

 

                let html = '';

                if (!isMe) {

                    html += `<div class="msg-sender">${escapeHtml(msg.sender)}</div>`;

                }

                html += `<div>${escapeHtml(msg.text)}</div>`;

               

                const timeStr = new Date(msg.timestamp || Date.now()).toLocaleTimeString([], { hour: '2-digit', minute: '2-digits' });

                html += `<div class="msg-time">${timeStr}</div>`;

 

                div.innerHTML = html;

                chatContainer.appendChild(div);

            });

 

            chatContainer.scrollTop = chatContainer.scrollHeight;

        }

 

        function escapeHtml(text) {

            const map = { '&': '&amp;', '<': '&lt;', '>': '&gt;', '"': '&quot;', "'": '&#039;' };

            return text.replace(/[&<>"']/g, m => map[m]);

        }

 

        messageInput.addEventListener('keypress', function (e) {

            if (e.key === 'Enter') {

                sendMessage();

            }

        });

 

        window.onload = initChat;

    </script>
